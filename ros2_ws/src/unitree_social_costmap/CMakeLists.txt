cmake_minimum_required(VERSION 3.8)
project(unitree_social_costmap)
find_package(ament_cmake REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(nav2_util REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_controller REQUIRED)
find_package(dwb_core REQUIRED)
find_package(nav_2d_utils REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(unitree_social_navigation_msgs REQUIRED)
find_package(behaviortree_cpp_v3 REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)

add_library(${PROJECT_NAME} SHARED src/people_layer.cpp)
add_library(unitree_point_components SHARED src/point_components.cpp)
target_compile_features(unitree_point_components PUBLIC cxx_std_17)
target_compile_options(unitree_point_components PRIVATE -O3)
install(TARGETS unitree_point_components LIBRARY DESTINATION lib)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_include_directories(${PROJECT_NAME} PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME} nav2_costmap_2d nav2_util pluginlib rclcpp tf2 tf2_ros unitree_social_navigation_msgs)
pluginlib_export_plugin_description_file(nav2_costmap_2d people_layer.xml)
install(TARGETS ${PROJECT_NAME} ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
add_library(unitree_replan_request_bt_node SHARED src/replan_requested.cpp)
target_compile_features(unitree_replan_request_bt_node PUBLIC cxx_std_17)
target_compile_definitions(unitree_replan_request_bt_node PRIVATE BT_PLUGIN_EXPORT)
target_include_directories(unitree_replan_request_bt_node PUBLIC include)
ament_target_dependencies(unitree_replan_request_bt_node behaviortree_cpp_v3 geometry_msgs nav_msgs rclcpp std_msgs)
install(TARGETS unitree_replan_request_bt_node LIBRARY DESTINATION lib)
add_library(unitree_validated_goal_checker SHARED src/validated_goal_checker.cpp)
target_compile_features(unitree_validated_goal_checker PUBLIC cxx_std_17)
target_include_directories(unitree_validated_goal_checker PUBLIC include)
ament_target_dependencies(unitree_validated_goal_checker nav2_controller nav2_core dwb_core nav_2d_utils geometry_msgs pluginlib)
pluginlib_export_plugin_description_file(nav2_core goal_checker.xml)
install(TARGETS unitree_validated_goal_checker LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_include_directories(include)
ament_export_libraries(${PROJECT_NAME})
ament_export_dependencies(nav2_costmap_2d nav2_util rclcpp tf2_ros unitree_social_navigation_msgs behaviortree_cpp_v3 geometry_msgs nav_msgs std_msgs)
ament_export_dependencies(nav2_controller nav2_core dwb_core)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_people_layer test/test_people_layer.cpp)
  target_link_libraries(test_people_layer ${PROJECT_NAME})
  ament_target_dependencies(test_people_layer nav2_costmap_2d nav2_util rclcpp tf2_ros unitree_social_navigation_msgs)
  ament_add_gtest(test_replan_requested test/test_replan_requested.cpp)
  target_link_libraries(test_replan_requested unitree_replan_request_bt_node)
  ament_target_dependencies(test_replan_requested behaviortree_cpp_v3 geometry_msgs nav_msgs rclcpp std_msgs)
  ament_add_gtest(test_validated_goal_checker test/test_validated_goal_checker.cpp)
  target_link_libraries(test_validated_goal_checker unitree_validated_goal_checker)
  ament_target_dependencies(test_validated_goal_checker nav2_controller nav2_core nav_2d_utils rclcpp tf2_ros pluginlib)
endif()
ament_package()
